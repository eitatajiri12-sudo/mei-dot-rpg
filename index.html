// ============================== // ALDIA - リーネ村 // PC版 基本ゲームシステム // ============================== const canvas = document.getElementById("cv"); const ctx = canvas.getContext("2d"); const WORLD = { width: 1900, height: 1450 }; // ------------------------------ // 主人公 // ------------------------------ const player = { x: 930, y: 900, speed: 185, direction: "down", moving: false, walkFrame: 0 }; // ------------------------------ // キー入力 // ------------------------------ const keys = { up: false, down: false, left: false, right: false }; const keyMap = { ArrowUp: "up", w: "up", ArrowDown: "down", s: "down", ArrowLeft: "left", a: "left", ArrowRight: "right", d: "right" }; window.addEventListener("keydown", (event) => { const key = event.key.toLowerCase(); if (keyMap[key]) { keys[keyMap[key]] = true; event.preventDefault(); } }); window.addEventListener("keyup", (event) => { const key = event.key.toLowerCase(); if (keyMap[key]) { keys[keyMap[key]] = false; } }); // ------------------------------ // NPC // ------------------------------ const NPCS = [ { name: "エレナ", sprite: "elena", x: 770, y: 470 }, { name: "薬師", sprite: "apothecary", x: 680, y: 720 }, { name: "武器屋", sprite: "smith", x: 1110, y: 755 }, { name: "村人", sprite: "farmer", x: 730, y: 1010 }, { name: "村人", sprite: "woman", x: 1040, y: 1000 }, { name: "子供", sprite: "child", x: 1180, y: 980 } ]; // ------------------------------ // 建物 // ------------------------------ const BUILDINGS = [ { name: "主人公の家", sprite: "house", x: 600, y: 130, width: 300, height: 302 }, { name: "教会", sprite: "church", x: 1040, y: 80, width: 330, height: 338 }, { name: "薬師の家", sprite: "apothecary_house", x: 350, y: 500, width: 290, height: 338 }, { name: "武器屋", sprite: "smith_house", x: 1180, y: 505, width: 285, height: 360 } ]; // ------------------------------ // 当たり判定 // ------------------------------ function isBlocked(x, y) { // マップ外 if ( x < 45 || y < 45 || x > WORLD.width - 45 || y > WORLD.height - 45 ) { return true; } // 川 if ( x > 1510 && !(y > 990 && y < 1240) ) { return true; } // 建物 for (const building of BUILDINGS) { if ( x > building.x + 20 && x < building.x + building.width - 20 && y > building.y + 65 && y < building.y + building.height - 8 ) { return true; } } return false; } // ------------------------------ // 主人公移動 // ------------------------------ function updatePlayer(deltaTime) { let dx = 0; let dy = 0; if (keys.left) dx -= 1; if (keys.right) dx += 1; if (keys.up) dy -= 1; if (keys.down) dy += 1; if (dx !== 0 || dy !== 0) { const length = Math.hypot(dx, dy); dx /= length; dy /= length; // 向き if (Math.abs(dx) > Math.abs(dy)) { player.direction = dx > 0 ? "right" : "left"; } else { player.direction = dy > 0 ? "down" : "up"; } const moveDistance = player.speed * deltaTime / 1000; const nextX = player.x + dx * moveDistance; const nextY = player.y + dy * moveDistance; if (!isBlocked(nextX, player.y)) { player.x = nextX; } if (!isBlocked(player.x, nextY)) { player.y = nextY; } player.moving = true; player.walkFrame += deltaTime * 0.04; } else { player.moving = false; } } // ------------------------------ // NPCを調べる // ------------------------------ function interact() { let nearestNPC = null; let nearestDistance = Infinity; for (const npc of NPCS) { const distance = Math.hypot( player.x - npc.x, player.y - npc.y ); if (distance < nearestDistance) { nearestDistance = distance; nearestNPC = npc; } } if ( nearestNPC && nearestDistance < 85 ) { talkToNPC(nearestNPC); } } // ------------------------------ // 会話 // ------------------------------ let storyStage = 0; function talkToNPC(npc) { if (npc.name === "エレナ") { if (storyStage === 0) { showDialog( "エレナ", "ごめんね……今日は少し身体が重くてね。\n" + "大丈夫。すぐ良くなるから。" ); storyStage = 1; setQuest( "薬師に母のことを相談しよう" ); } } else if (npc.name === "薬師") { if (storyStage >= 1) { showDialog( "薬師", "ただの風邪じゃなさそうだね。\n" + "「月白草」があれば薬を作れる。\n" + "北のルナ洞窟に生えているはずだよ。" ); storyStage = 2; setQuest( "武器屋で洞窟へ行く準備をしよう" ); } } else if (npc.name === "武器屋") { if (storyStage >= 2) { showDialog( "武器屋", "洞窟へ行くだと？\n" + "……母ちゃんのためか。\n\n" + "この古びた剣を貸してやる。\n" + "生きて返しに来いよ。" ); storyStage = 3; setQuest( "北のルナ洞窟へ向かおう" ); } } } // ------------------------------ // SPACE / ENTER // ------------------------------ window.addEventListener( "keydown", (event) => { if ( event.key === " " || event.key === "Enter" ) { if (!event.repeat) { interact(); } } } ); // ------------------------------ // カメラ // ------------------------------ function getCamera() { let cameraX = player.x - canvas.width / 2; let cameraY = player.y - canvas.height / 2; cameraX = Math.max( 0, Math.min( WORLD.width - canvas.width, cameraX ) ); cameraY = Math.max( 0, Math.min( WORLD.height - canvas.height, cameraY ) ); return { x: cameraX, y: cameraY }; } // ------------------------------ // メインループ // ------------------------------ let previousTime = performance.now(); function gameLoop(currentTime) { const deltaTime = Math.min( 34, currentTime - previousTime ); previousTime = currentTime; updatePlayer(deltaTime); drawGame(); requestAnimationFrame( gameLoop ); } requestAnimationFrame( gameLoop );
